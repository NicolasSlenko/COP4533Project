\section{Algorithm Design and Analysis}

\subsection{Algorithm 1 (ProblemS1)}

\subsubsection{Description}

\noindent\textbf{Greedy strategy.} We start at $0$ and jump as far forward as possible without passing $n$. Costs never increase toward the right, so the farthest reachable panel costs no more than an earlier one. We record each landing and its cost until reaching $n$.

\medskip
\noindent\textbf{Example.} For instance, let $n=8$, $k=3$, and
\[
c=[20,18,16,14,12,10,8,6].
\]
The route is $0\to3\to6\to8$, with total cost
\[
c_3+c_6+c_8=16+10+6=32.
\]

\begin{algorithm}[H]
\caption{Greedy Route for ProblemS1}
\label{alg:s1}
\begin{algorithmic}[1]
\Procedure{PanelRoute}{$n,k,[c_1,\ldots,c_n]$}
    \State $S \gets [0]$ \Comment{route list starting at the dock}
    \State $E \gets 0$ \Comment{total landing cost}
    \While{the last element of $S$ is less than $n$}
        \State $j \gets \min(n,\text{last element of }S+k)$
        \State $E \gets E+c_j$
        \State Append $j$ to $S$
    \EndWhile
    \State \Return $E$, $S$ excluding the first element of $S$
\EndProcedure
\end{algorithmic}
\end{algorithm}

\subsubsection{Correctness Proof}

\noindent\textbf{Theorem.} Algorithm 1 returns a valid cheapest route for ProblemS1.

\medskip
\noindent\textbf{Proof.} [By induction]
Let $G=\{g_1,\ldots,g_m\}$ be the greedy route and let $O=\{o_1,\ldots,o_t\}$ be any valid route, with indices listed in increasing order. We initially set $g_0=o_0=0$.

\begin{enumerate}
    \item Greedy jumps $k$ positions each time, shortening its final jump to land on $n$. Thus, $m=\lceil n/k\rceil$ and $g_r=\min(rk,n)$ for $0\le r\le m$. Every jump is legal, so $G$ is valid.

    \item The $t$ jumps in $O$ cover at most $tk$ positions. Since $O$ reaches $n$, $t\ge\lceil n/k\rceil=m$. Hence, $o_r$ exists for every $0\le r\le m$.

    \item Let $r$ denote the number of jumps taken, so $o_r$ and $g_r$ are the panels reached after $r$ jumps. We prove by induction on $r$ that $o_r\le g_r$. The base case is $o_0=g_0=0$. For the inductive step, suppose $o_r\le g_r$ for some $r<m$. The next jump in $O$ has length at most $k$ and cannot pass $n$, so
    \[
    o_{r+1}\le\min(o_r+k,n)\le\min(g_r+k,n)=g_{r+1}.
    \]
    Therefore, the claim holds after every jump through $m$.

    \item Since costs never increase toward the right, $o_r\le g_r$ implies $c_{g_r}\le c_{o_r}$ for $1\le r\le m$. Summing gives
    \[
    \sum_{r=1}^{m}c_{g_r}\le\sum_{r=1}^{m}c_{o_r}\le\sum_{r=1}^{t}c_{o_r},
    \]
    where the last inequality follows because any extra landings in $O$ have positive cost.
\end{enumerate}

Thus, $G$ is valid and costs no more than any valid route $O$. Hence, the greedy algorithm is optimal. \hfill$\square$

\subsubsection{Runtime Analysis}

Let $m=\lceil n/k\rceil$. The loop makes $m$ jumps with constant time work per jump. Copying the returned route takes $\Theta(m)$ time. Together, these give
\[
\Theta(m)+\Theta(m)=\Theta(m).
\]
The function's worst case is $\Theta(n)$ when $k=1$. Reading all $n$ costs makes the complete program's running time
\[
\Theta(n)+\Theta(m)=\Theta(n).
\]
The route uses $\Theta(m)$ extra space.

\subsection{Algorithm 2 (ProblemS2)}

\subsubsection{Description}

\noindent\textbf{Greedy strategy.} Let $p$ be the peak panel. We try each jump $a\to a+k$ with $a\le p\le a+k$. We build the left side backward from $a$ in jumps of $k$ toward $0$, then reverse it. We then build the right side by jumping as far forward as possible from $a+k$ to $n$ and then sum each complete route's cost and keep the cheapest.

Costs rise before the peak and fall after it, which allows ties away from the peak. Backward jumps favor cheaper panels on the left, and forward jumps favor cheaper panels on the right. The crossing has the option of starting or ending at the peak.

\medskip
\noindent\textbf{Example.} Let $n=10$, $k=4$, and
\[
c=[1,2,4,9,12,10,6,4,3,1].
\]
The peak is $p=5$. The possible crossing jumps and completed routes are:
\begin{center}
\begin{tabular}{ccl}
\textbf{Crossing} & \textbf{Visited panels} & \textbf{Energy} \\
\hline
$1\to5$ & $[1,5,9,10]$ & $1+12+3+1=17$ \\
$2\to6$ & $[2,6,10]$ & $2+10+1=13$ \\
$3\to7$ & $[3,7,10]$ & $4+6+1=11$ \\
$4\to8$ & $[4,8,10]$ & $9+4+1=14$ \\
$5\to9$ & $[1,5,9,10]$ & $1+12+3+1=17$
\end{tabular}
\end{center}
The algorithm returns $[3,7,10]$ with total energy $11$.

\begin{algorithm}[H]
\caption{Greedy Completions for ProblemS2}
\label{alg:s2}
\begin{algorithmic}[1]
\Procedure{PeakRoute}{$n,k,[c_1,\ldots,c_n]$}
    \State $p\gets$ index of the maximum landing cost
    \State $E^*\gets\infty$, $S^*\gets []$ \Comment{best cost and route found so far}
    \For{$a=\max(0,p-k)$ to $\min(p,n-k)$}
        \State $S\gets []$, $E\gets0$ \Comment{candidate route and its energy cost}
        \State $j\gets a$
        \While{$j>0$}
            \State Append $j$ to $S$
            \State $j\gets\max(0,j-k)$
        \EndWhile
        \State Reverse $S$
        \State $j\gets a+k$
        \State Append $j$ to $S$
        \While{$j\ne n$}
            \State $j\gets\min(n,j+k)$
            \State Append $j$ to $S$
        \EndWhile
        \For{each panel $i$ in $S$}
            \State $E\gets E+c_i$
        \EndFor
        \If{$E<E^*$}
            \State $E^*\gets E$, $S^*\gets S$
        \EndIf
    \EndFor
    \State \Return $E^*,S^*$
\EndProcedure
\end{algorithmic}
\end{algorithm}

\subsubsection{Correctness Proof}

\noindent\textbf{Theorem.} Algorithm 2 returns a valid cheapest route for ProblemS2.

\medskip
\noindent\textbf{Proof.} [Direct proof]
Let $G=\{g_1,\ldots,g_m\}$ be the returned route and let $O=\{o_1,\ldots,o_t\}$ be any valid route, with the indices listed in increasing order and $o_0=0$. Let $E(G)$ and $E(O)$ denote their energy costs.

\begin{enumerate}
    \item Let $j$ be the first landing number with $o_j\ge p$, and set $x=o_{j-1}$, $y=o_j$. Since $O$ reaches $n$, this crossing exists, with $x<p\le y$ and $1\le y-x\le k$.

    \item Let $L(a)$ denote the backward greedy completion cost from $a\le p$ to $0$, excluding $0$, and $R(b)$ the forward greedy completion cost from $b\ge p$ to $n$, including $b$.

    \item Costs never increase on the right, so the S1 proof applies. Greedy stays at least as far right after the same number of jumps and uses the fewest jumps. Thus, $R(b)$ is the cheapest continuation.

    \item Costs never decrease toward the peak on the left. Let $r$ count backward jumps from $a$. The positive greedy landing $a-rk$ lies no farther right than another route's corresponding landing, so it costs no more. Greedy uses the fewest landings, and extra landings have positive cost. Thus, $L(a)$ is cheapest, giving
    \[
    L(x)+R(y)\le E(O).
    \]

    \item Moving $a$ left moves each matched backward landing left and cannot add landings, so $L(a)$ cannot increase. Similarly, moving $b$ right moves each matched forward landing right and cannot add landings, so $R(b)$ cannot increase.

    \item The original crossing may be shorter than $k$. Let $x'$ and $y'$ be the endpoints of a replacement crossing. To keep $y$ fixed and make the gap $k$, we would start at $y-k$. If this lies before $0$, we instead start at $0$ and end at $k$. Both cases are expressed by
    \[
    x'=\max(0,y-k),\qquad y'=x'+k=\max(y,k).
    \]
    The maximum keeps $x'$ inside the row. Since $y-x\le k$, we have $y-k\le x$, so $x'\le x$. Also, $y'\ge y$ and $y'\le n$, since $y,k\le n$. Thus, the endpoints move outward, the crossing still spans the peak, and $y'-x'=k$.

    \item Moving the endpoints outward cannot increase either completion cost, as shown in step 5. Thus,
    \[
    L(x')+R(y')\le L(x)+R(y)\le E(O).
    \]
    Any shorter crossing can therefore be replaced by one of length $k$ without increasing the complete route's cost. This justifies considering only crossings of length $k$.

    \item A candidate starting at $a$ must satisfy $a\ge0$ and $a+k\le n$ to stay inside the row. It must also satisfy $a\le p\le a+k$ to span the peak. Combining these conditions gives
    \[
    \max(0,p-k)\le a\le\min(p,n-k).
    \]
    The replacement crossing satisfies these conditions, so the algorithm checks it. Since it returns the cheapest candidate,
    \[
    E(G)\le L(x')+R(y')\le E(O).
    \]
\end{enumerate}

Every candidate starts at $0$, ends at $n$, and uses jumps of lengths $1$ through $k$. Since $O$ was arbitrary, $G$ is valid and costs no more than any valid route. Hence, the algorithm is optimal. \hfill$\square$

\subsubsection{Runtime Analysis}

Finding the peak takes $\Theta(n)$ time. There are at most $k+1$ candidates, each with $O(n/k+1)$ landings. Building, reversing, and summing each route takes time proportional to its length. Since $1\le k\le n$, all candidates take $O((k+1)(n/k+1))=O(n)$ time. Adding the peak scan gives
\[
\Theta(n)+O\bigl((k+1)(n/k+1)\bigr)
=\Theta(n)+O(n)
=\Theta(n).
\]
The current and best routes use $O(n/k+1)$ extra space.

\section{Experimental Comparative Study}

\subsection{Experimental Setup}
We used Python version 3.12.14 on Windows with a random seed of $4533$. We generated the S1 and S2 datasets using Python's \texttt{random} module.

\begin{description}
    \item[ProblemS1:] We sampled $n$ costs uniformly from $[1,99999]$ and sorted them in decreasing order.
    \item[ProblemS2:] We chose an interior peak position $p$ and a peak cost from $[2,99999]$. We sampled both sides below that cost, then sorted the left side in increasing order and the right side in decreasing order.
    \item[Input sizes:] We tested $n\in\{20000,40000,60000,80000,99999\}$ with $k=50$. We checked every generated input against the problem restrictions and saved one input file per configuration.
    \item[Timing:] We used \texttt{time.perf\_counter()} to measure only the algorithm function. We generated and validated inputs before timing. After a warmup call, we averaged a batch of 50 calls for Program 1 or 5 calls for Program 2. We reported the mean and sample standard deviation across 20 random inputs per configuration and saved all 400 trial timings.
    \item[Fitted curves:] We fitted $T_1=c_1(n/k)$ and $T_2=c_2n$ by least squares with zero intercept for an expected comparison. The fitted constants came from the measurements.
\end{description}

\subsection{Correctness Testing}
We tested 100,000 random inputs per program using seed $4533$. For S1, $1\le n\le14$. For S2, $3\le n\le14$ with a unique interior peak. Each test chose $1\le k\le n$. Ordinary costs were at most $30$, and S2 peak costs were at most $35$. We checked jump lengths, arrival at $n$, and the sum of landing costs. We also compared each returned cost with an independent $O(nk)$ dynamic programming reference. All tests and the assignment examples passed.

\subsection{Observations}
\begin{enumerate}
    \item With $k=50$, both of the measured functions grow roughly linearly with $n$.
    \item The error bars show the spread across trials. Some measurements vary noticeably, especially Program 1 at small runtimes. 
    \item As $k$ increases from $10$ to $200$, Program 1 drops from about $1.21$ ms to $0.060$ ms. Larger jumps reduce its $\lceil n/k\rceil$ loop iterations.
    \item Over the tested values of $k$, Program 2 stays in the  $13$ ms range. This supports the analysis that $O(k)$ candidates with $O(n/k)$ landings give $O(n)$ work in this range, plus the $\Theta(n)$ peak scan.
\end{enumerate}

\section{Conclusion}

\subsection{Summary of Results}
We designed, proved, and implemented greedy algorithms for both special cases. For \textsc{ProblemS1}, jumping as far as possible without passing $n$ is optimal. For \textsc{ProblemS2}, trying every crossing of length $k$ across the peak, completed greedily on both sides, is optimal. Both complete programs run in $\Theta(n)$ time. The counterexamples show why the S1 rule fails on the other cost models. The experiments support the predicted growth in $n$ and $k$.
